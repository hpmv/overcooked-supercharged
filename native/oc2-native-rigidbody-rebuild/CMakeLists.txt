cmake_minimum_required(VERSION 3.15)
project(Oc2NativeRigidbodyRebuild LANGUAGES CXX)
add_library(Oc2NativeRigidbodyRebuild SHARED Oc2NativeRigidbodyRebuild.cpp)
target_compile_features(Oc2NativeRigidbodyRebuild PRIVATE cxx_std_14)
target_compile_definitions(Oc2NativeRigidbodyRebuild PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(Oc2NativeRigidbodyRebuild PRIVATE /W4 /WX /GS- /guard:cf-)
set_target_properties(Oc2NativeRigidbodyRebuild PROPERTIES OUTPUT_NAME "Oc2NativeRigidbodyRebuild")
option(OC2_BUILD_CONTACT_HISTORY_HARNESS "Build caller-owned contact-pool snapshot tests" OFF)
if(OC2_BUILD_CONTACT_HISTORY_HARNESS)
    add_executable(Oc2NativeRigidbodyRebuildHistoryHarness
        Oc2NativeRigidbodyRebuildHistoryHarness.cpp)
    target_compile_features(Oc2NativeRigidbodyRebuildHistoryHarness PRIVATE cxx_std_14)
    target_compile_definitions(Oc2NativeRigidbodyRebuildHistoryHarness PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_compile_options(Oc2NativeRigidbodyRebuildHistoryHarness PRIVATE /W4 /WX)
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Oc2NativeRigidbodyRebuild must be built for Win32/x86")
endif()
